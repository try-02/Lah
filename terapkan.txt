        internal fun petakanPesanRamah(e: GagalSftp, profil: ProfilSftp? = null): Pair<String, PenyebabGalat> = when (e) {
            is GagalSftp.TidakAda -> (e.message ?: "Tidak ditemukan") to PenyebabGalat.TIDAK_DITEMUKAN
            is GagalSftp.SudahAda -> (e.message ?: "Target sudah ada") to PenyebabGalat.KONFLIK
            is GagalSftp.IzinDitolak,
            is GagalSftp.AutentikasiGagal,
            is GagalSftp.RahasiaHilang,
            is GagalSftp.HostBelumDipercaya,
            is GagalSftp.HostBerubah -> (e.message ?: "Izin ditolak") to PenyebabGalat.IZIN_DITOLAK
            is GagalSftp.AdalahFolder,
            is GagalSftp.TautanBerputar -> (e.message ?: "Input tidak valid") to PenyebabGalat.VALIDASI
            is GagalSftp.Jaringan -> {
                val sebab = e.cause
                val host = profil?.host
                val port = profil?.port
                val tambahan = if (host != null && port != null && (host == "127.0.0.1" || host == "localhost") &&
                    (sebab is ConnectException || e.message.orEmpty().contains("refused", ignoreCase = true))
                ) {
                    "Koneksi ke $host:$port ditolak. Pastikan `sshd` berjalan di Termux."
                } else if (host != null && sebab is UnknownHostException) {
                    "Host '$host' tidak ditemukan."
                } else {
                    e.message ?: "Kegagalan jaringan SFTP"
                }
                tambahan to PenyebabGalat.TIDAK_DIKETAHUI
            }
            is GagalSftp.Terputus,
            is GagalSftp.Lainnya -> (e.message ?: "Kesalahan SFTP") to PenyebabGalat.TIDAK_DIKETAHUI
        }

Dan untuk SqlDelightPenyimpananRahasiaSftp:

@OptIn(ExperimentalTime::class)
class SqlDelightPenyimpananRahasiaSftp(
    private val database: DatabaseAplikasi,
    private val pengelola: PengelolaRahasiaSftp = PengelolaRahasiaSftp(),
    private val ioDispatcher: CoroutineDispatcher = Dispatchers.IO,
    private val clock: Clock = Clock.System
) : PenyimpananRahasiaSftp {

    override suspend fun simpan(idProfil: String, rahasia: RahasiaSftp) {
        withContext(ioDispatcher) {
            val tersegel = pengelola.segel(rahasia, idProfil)
            database.preferensiEntityQueries.simpanNilai(
                kunci = "$PREFIX_RAHASIA$idProfil",
                nilai = tersegel,
                waktu_diubah = clock.now().toEpochMilliseconds()
            )
        }
    }

    override suspend fun ambil(idProfil: String): RahasiaSftp? = withContext(ioDispatcher) {
        val mentah = database.preferensiEntityQueries.ambilNilai("$PREFIX_RAHASIA$idProfil").executeAsOneOrNull()
            ?: return@withContext null
        pengelola.buka(mentah, idProfil)
    }

    override suspend fun hapus(idProfil: String) {
        withContext(ioDispatcher) {
            database.preferensiEntityQueries.hapusKunci("$PREFIX_RAHASIA$idProfil")
        }
    }

    override suspend fun ada(idProfil: String): Boolean = withContext(ioDispatcher) {
        database.preferensiEntityQueries.ambilNilai("$PREFIX_RAHASIA$idProfil").executeAsOneOrNull() != null
    }
}
